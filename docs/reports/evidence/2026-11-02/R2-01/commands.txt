## 2. JSON Schema 校验（核心验收，实跑）
# 2.1 schema 自身必须是合法 draft 2020-12 schema
PYTHONUTF8=1 python -c "import json;from jsonschema import Draft202012Validator;Draft202012Validator.check_schema(json.load(open('docs/contracts/event-bus.schema.json',encoding='utf-8')));print('check_schema: OK  meta=%s'%Draft202012Validator.META_SCHEMA['\$id'])"
# -> check_schema: OK  meta=https://json-schema.org/draft/2020-12/schema

# 2.2 逐条校验 21 个示例（14 正向 + 7 反向）。exit 0 = 合法，exit 1 = 被拒绝。
for f in docs/contracts/testdata/event-bus/*.json; do
  PYTHONUTF8=1 python -m jsonschema -o plain -i "$f" docs/contracts/event-bus.schema.json >/dev/null 2>/tmp/e.txt
  echo "$(basename "$f") exit=$?"
done
# 完整输出（含每条拒绝理由）见 logs/schema-validation.txt

# 2.3 Windows 编码坑：不带 PYTHONUTF8=1 时 CLI 以 gbk 打开含中文的 schema 直接崩溃
PYTHONUTF8= PYTHONIOENCODING= python -m jsonschema -o plain \
  -i docs/contracts/testdata/event-bus/run_finished.json docs/contracts/event-bus.schema.json
# -> exit=1  UnicodeDecodeError: 'gbk' codec can't decode byte 0x80 in position 232

# 2.4 node 交叉解析（证明 schema 与全部示例都是可解析 JSON；node 必须用绝对路径）
/e/node_js/node.exe -e 'const fs=require("fs");JSON.parse(fs.readFileSync("docs/contracts/event-bus.schema.json","utf8"));const d="docs/contracts/testdata/event-bus";const n=fs.readdirSync(d).filter(f=>f.endsWith(".json")).length;fs.readdirSync(d).forEach(f=>JSON.parse(fs.readFileSync(d+"/"+f,"utf8")));console.log("schema + "+n+" examples parsed by node: ok ("+process.version+")")'
# -> schema + 21 examples parsed by node: ok (v20.12.2)

# 2.5 allowlist / 必填字段与代码和计划的一致性交叉检查
